cmake_minimum_required(VERSION 3.16)
project(sqlite_littlefs_vfs VERSION 0.1.0 LANGUAGES C)
set(_tests_default OFF)
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  set(_tests_default ON)
endif()
option(SQLITE_LITTLEFS_BUILD_TESTS "Build host qualification tests" ${_tests_default})
if(SQLITE_LITTLEFS_BUILD_TESTS)
  include(CTest)
endif()
option(SQLITE_LITTLEFS_SANITIZE "Run host tests with AddressSanitizer and UBSan" OFF)
if(SQLITE_LITTLEFS_SANITIZE)
  set(_crash_timeout_default 3600)
else()
  set(_crash_timeout_default 900)
endif()
set(SQLITE_LITTLEFS_CRASH_TIMEOUT "${_crash_timeout_default}" CACHE STRING "Crash/recovery test timeout in seconds")
if(NOT SQLITE_LITTLEFS_CRASH_TIMEOUT MATCHES "^[1-9][0-9]*$")
  message(FATAL_ERROR "SQLITE_LITTLEFS_CRASH_TIMEOUT must be a positive number of seconds")
endif()
set(SQLITE_LITTLEFS_SQLITE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/sqlite" CACHE PATH "Directory containing sqlite3.h")
set(SQLITE_LITTLEFS_LITTLEFS_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/littlefs" CACHE PATH "Directory containing lfs.h")
foreach(header "${SQLITE_LITTLEFS_SQLITE_INCLUDE_DIR}/sqlite3.h" "${SQLITE_LITTLEFS_LITTLEFS_INCLUDE_DIR}/lfs.h")
  if(NOT EXISTS "${header}")
    message(FATAL_ERROR "Missing ${header}; run python3 tools/fetch_deps.py or set dependency include directories")
  endif()
endforeach()

add_library(sqlite_littlefs_vfs STATIC src/sqlite_littlefs_vfs.c)
add_library(sqlite_littlefs_vfs::sqlite_littlefs_vfs ALIAS sqlite_littlefs_vfs)
target_compile_features(sqlite_littlefs_vfs PUBLIC c_std_11)
target_include_directories(sqlite_littlefs_vfs PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}/include"
  "${SQLITE_LITTLEFS_SQLITE_INCLUDE_DIR}"
  "${SQLITE_LITTLEFS_LITTLEFS_INCLUDE_DIR}")
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(sqlite_littlefs_vfs PRIVATE -Wall -Wextra -Werror)
endif()

if(SQLITE_LITTLEFS_BUILD_TESTS AND BUILD_TESTING)
  if(NOT UNIX)
    message(FATAL_ERROR "The fork-based crash harness currently requires macOS or Linux; the VFS itself is portable")
  endif()
  # Keep LittleFS struct-layout options identical across the VFS, upstream
  # LittleFS, and its callers. The production consumer supplies its own flags.
  target_compile_definitions(sqlite_littlefs_vfs PUBLIC LFS_NO_MALLOC LFS_MULTIVERSION LFS_NO_DEBUG LFS_NO_WARN LFS_NO_ERROR)
  add_library(test_littlefs STATIC vendor/littlefs/lfs.c vendor/littlefs/lfs_util.c)
  target_include_directories(test_littlefs PUBLIC vendor/littlefs)
  # Keep upstream invariants enabled even in optimized qualification runs.
  target_compile_options(test_littlefs PRIVATE -UNDEBUG)
  target_compile_definitions(test_littlefs PUBLIC LFS_NO_MALLOC LFS_MULTIVERSION LFS_NO_DEBUG LFS_NO_WARN LFS_NO_ERROR)
  add_library(test_sqlite STATIC vendor/sqlite/sqlite3.c tests/sqlite_port.c)
  target_include_directories(test_sqlite PUBLIC vendor/sqlite tests)
  target_compile_options(test_sqlite PUBLIC -include "${CMAKE_CURRENT_SOURCE_DIR}/tests/sqlite_config.h")
  target_link_libraries(test_sqlite PUBLIC m)
  add_library(example_mounted OBJECT examples/mounted.c)
  target_link_libraries(example_mounted PRIVATE sqlite_littlefs_vfs)
  add_executable(host_test tests/host_test.c)
  target_link_libraries(host_test PRIVATE sqlite_littlefs_vfs test_littlefs test_sqlite)
  target_compile_options(host_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME host_api COMMAND host_test --quick)
  add_test(NAME crash_recovery COMMAND host_test --crash)
  set_tests_properties(host_api PROPERTIES TIMEOUT 60)
  set_tests_properties(crash_recovery PROPERTIES TIMEOUT "${SQLITE_LITTLEFS_CRASH_TIMEOUT}")
  # A less-trimmed desktop build catches assumptions hidden by embedded
  # OMIT_* options, including WAL probes and extension loader callbacks.
  add_library(stock_sqlite STATIC vendor/sqlite/sqlite3.c)
  target_include_directories(stock_sqlite PUBLIC vendor/sqlite)
  target_compile_definitions(stock_sqlite PRIVATE SQLITE_ENABLE_MEMSYS5=1 SQLITE_ZERO_MALLOC=1 SQLITE_THREADSAFE=0 SQLITE_TEMP_STORE=3 SQLITE_DEFAULT_LOOKASIDE=0,0)
  target_link_libraries(stock_sqlite PUBLIC m ${CMAKE_DL_LIBS})
  add_executable(stock_host_test tests/host_test.c)
  target_link_libraries(stock_host_test PRIVATE sqlite_littlefs_vfs test_littlefs stock_sqlite)
  target_compile_options(stock_host_test PRIVATE -Wall -Wextra -Werror)
  add_test(NAME stock_sqlite_api COMMAND stock_host_test --quick)
  set_tests_properties(stock_sqlite_api PROPERTIES TIMEOUT 300)
  if(SQLITE_LITTLEFS_SANITIZE)
    foreach(target sqlite_littlefs_vfs test_littlefs test_sqlite host_test stock_sqlite stock_host_test)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endforeach()
  endif()
endif()
